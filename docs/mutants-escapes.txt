# Documented equivalent mutants, one regex per line (Rust `regex` syntax, matched anywhere in a
# mutant's name such as `crates/ltx/src/ids.rs:61:9: replace | with ^ in <impl FromStr for Txid>::from_str`).
# `just mutants-diff` passes each line to cargo-mutants as --exclude-re, so a matching mutant is
# not run and cannot fail a PR. Every line carries the reason, and the reason lives in
# docs/mutants.md ("Equivalent" tables); add the line and the table row in one commit.
# Do not put line numbers in a regex: they move.

# ltx checksum.rs DbChecksum::toggle: both operands carry bit 63, so OR and XOR agree.
replace \| with \^ in .*DbChecksum::toggle
# ltx ids.rs Txid::from_str: the shifted value has a clear low nibble, so OR and XOR agree.
replace \| with \^ in .*Txid.*from_str
# ltx index.rs put_uvarint: `v & 0x7f` has bit 7 clear, so OR and XOR of 0x80 agree.
replace \| with \^ in put_uvarint
# core layout.rs FileKey::from_str: same shift-and-digit argument as Txid.
replace \| with \^ in .*FileKey.*from_str
# core plan.rs extend: `preference` ends in the file's own range, so two distinct files never tie and `>` vs `>=` cannot pick differently.
replace > with >= in extend
# core plan.rs Sweep::next: `sweep_order` includes the key, unique per file, so `<` vs `<=` never meets a tie.
replace < with <= in <impl Iterator for Sweep<'a>>::next
# core witness.rs decode: every file the boot pass decodes already passed sweep_staged or the listing's read-back, both of which require the header to name the key.
replace match guard verified.header\(\).txids\(\) == key.range\(\) with true in decode
# core pages.rs seeded: a quiet mark at or below the confirmed transaction is trimmed before use (`observe_lag`), so `>` vs `>=` at it cannot be told apart.
replace > with >= in seeded
# core actor/lag.rs observe_lag: a quiet mark equal to the replicated transaction is outside the range `only_quiet_pending` reads, so keeping it changes nothing.
replace > with >= in Actor::observe_lag
# core store.rs on_page: an unrecognized name never equals a key name, so `>` and `>=` at the cursor agree.
replace > with >= in on_page
# core store.rs ObjectStore::batch_width: the default already returns one.
replace ObjectStore::batch_width -> NonZeroUsize with 1.try_into\(\).unwrap\(\)
# core retain.rs past_snapshot: a file ending at the snapshot neither opens a hole nor moves the reach.
replace > with >= in past_snapshot
# core retain.rs latest_unplannable: a Latest plan always reaches the newest transaction; the guard is kept as a check.
replace match guard chain.reaches\(\) == newest with true in latest_unplannable
# core retain.rs pinned_chain: a Txid plan exists only when it reaches the target; the guard is kept as a check.
replace match guard chain.reaches\(\) == restore_point with true in pinned_chain
# core verify/against.rs compare: both sides hold every page up to their commit, so equal pages mean equal commits.
replace match guard commit == live.commit.get\(\) with true in compare
# core verify/dry_run.rs DigestSink::reset: a reset is followed by a chain that rewrites or truncates every page.
dry_run\.rs:.*DigestSink.*::reset .* with Ok
# core verify/dry_run.rs Counting::delete: a dry-run restore never deletes.
dry_run\.rs:.*Counting.*::delete .* with Ok
# core actor.rs prune_unit: a whole page taken implies a next page, since the newest staged file is listed above the bound; past one page the mutant costs one extra listing.
replace && with \|\| in Actor::prune_unit
# core actor.rs arm_takeover_below: a listing ending at the record's transaction confirms everything the record covers, so arming then reaches no publish (docs/mutants.md, takeover spend).
replace < with <= in Actor::arm_takeover_below
# wal testing.rs page: the test-data generator's splitmix64 mixing steps; a weaker mixer still gives deterministic pages that differ by seed and page number, and changing it would change every recorded simulator history.
wal/src/testing\.rs:.* in page$
# store-fs fsync wrappers: replacing an fsync with Ok(()) shows only after a power loss; the crash tests in docs/planted-bugs.toml check the store makes these calls, and the LazyFS rig checks the real syscalls.
engine_journal\.rs:.*FsJournalFile>::sync(_dir)? -> .* with Ok\(\(\)\)
filesystem\.rs:.*for RealFs>::sync_dir -> .* with Ok\(\(\)\)
filesystem\.rs:.*for File>::sync_all -> .* with Ok\(\(\)\)
# store-fs engine_journal.rs read_slot: besides end of file, a read of an open journal fails only on a device fault no test can produce.
replace match guard err.kind\(\) == ErrorKind::UnexpectedEof with true in <impl JournalFile for FsJournalFile>::read_slot
